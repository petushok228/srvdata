Тема 1. Класс Human

 class Human
    {
        // описание полей
        private string name; //Защищенное поле имя
        private string secondname;
        private string lastname;
        private int age;//Защищенное поле возраст
                        // описание свойств
                        //Свойства объекта – это средства доступа к полям, позволяющие изменять его данные и выполнять его код. 
        public string Name
        {
            set
            {
                name = value;
            }
        }
        public string SecondName
        {
            set
            {
                secondname = value;
            }
        }
        public string LastName
        {
            set
            {
                lastname = value;
            }
        }

        public int Age
        {
            get
            {
                return age;
            }
            set
            {
                age = value;
            }
        }
        //Метод возвращаюций полное фамилия, имя и отчество
        public string FullName
        {
            get
            {
                return lastname + " " + name + " " + secondname;
            }
        }
        public Human()  //конструктор класса Human без ввода полей
        { }

        //конструктор класса Human
        public Human(string lastname, string name, string secondname, int age)
        {   //this представляет ссылку на текущий экземпляр/объект класса
            this.lastname = lastname;
            this.name = name;
            this.secondname = secondname;
            this.age = age;
        }
        // описание метода вывод информации о человеке
        public void Info()
        {
            string s;
            switch (age % 10) //используется деление возраста и определение остатка
            {       //если остаток 1 добавляем год  
                case 1:
                    s = " год";
                    break;
                case 2:  //  если остаток 2 (32, 33, 34) добавляем слово года
                case 3:
                case 4:
                    s = " года";
                    break;
                default:   //  если остаток 5 или 6 или 7 (35, 36, 37) добавляем слово лет
                    s = " лет";
                    break;
            } // сформированная строка
            Console.WriteLine(lastname + " " + name + " " + secondname + "," + age + s);
        }
    }
