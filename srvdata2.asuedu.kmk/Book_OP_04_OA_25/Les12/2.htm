Постановка задачи. Напечатать ряд из 20 повторяющихся чисел 20 в виде:
а) строки: 20 20 20 20 20 20 20 20 20 20
b) таблицы 4х5
20 20 20 20
20 20 20 20
20 20 20 20
20 20 20 20
20 20 20 20
1. Ввести переменную для числа 20 digA = 20
2. Вывести число digA в строку
3. Вывести число digA в таблицу по 5 чисел в строке, 4 строк

Пример реализации в Pascal

program Ur11;

{$APPTYPE CONSOLE}
//Урок 11. Циклы: с постусловием, с предусловием, с параметром.
{$R *.res}
// Задача 2. Напечатать ряд из 10 повторяющихся чисел 20 в виде: 20 20 20 20 20 20 20 20 20 20
uses
  System.SysUtils;
 //1. Ввести переменную для числа 20  digA = 20
 var
      digA: byte = 20;
      i: byte;  //объявили переменную для счетныый цикл
      j: byte;  // объявили переменную для Цикла while с предусловием
      L: byte;  // объявили переменную для Цикла repeat  с ПОСТУСЛОВИЕМ

begin
  WriteLn('Работа 11 студент Иванов А.');
  for I := 1 to 20 do   //цикл повторяется 10 раз  счетныый цикл
   begin
     Write(digA,' ');
     {
      if i = 4 then  WriteLn;
      if i = 8 then  WriteLn;
      if i = 12 then  WriteLn;
      if i = 16 then  WriteLn;
      }
      case i of
      4:  WriteLn;
      8:  WriteLn;
      12: WriteLn;
      16: WriteLn;
      end;
   end;
    WriteLn;
    WriteLn('Цикл while с предусловием');
   //Цикл while с предусловием    while(i > 0) { тело }
   j:=20; //определили количество итераций
    while(j > 0) do { тело }
       begin
         Write(digA,' ');
           case j of
              5:  WriteLn;
              9:  WriteLn;
              13: WriteLn;
              17: WriteLn;
            end;
         j:=j-1;  //уменьшаем количество итераций на 1
       end;

    //Цикл repeat  с ПОСТУСЛОВИЕМ    repeat   оператор1;  . . .;  операторN
   //until условие {до тех пор, пока условие не станет истинным}
   WriteLn;
    WriteLn('Цикл repeat с ПОСТУСЛОВИЕМ');
   L:=20;   //определили количество итераций
     repeat
          Write(digA,' ');
          l:=l-1;  //уменьшаем количество итераций на 1
          case l of
              4:  WriteLn;
              8:  WriteLn;
              12: WriteLn;
              16: WriteLn;
            end;
     until (L <= 0);



  Readln; //точка останова
end.

Пример реализации в C#

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
//Задача 2. Напечатать ряд из 20 повторяющихся чисел 20 в виде: 20 20 20 20 20 20 20 20 20 20
namespace Ur11_ABD
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("Урок 11 Циклы  работа студента Иванова А");
            //Ввести переменную   числа digA = 20 20 раз
            int digA = 20;
            // 2.Вывести число digA в строку
            for (int i = 0; i < 20; i++) 
            {
                Console.Write($"{digA}  ");
                if (i == 3) { Console.WriteLine(); }
                if (i == 7)  {Console.WriteLine();}
                if (i == 11) { Console.WriteLine();}
                if (i == 15) { Console.WriteLine(); }
            }

            Console.WriteLine();
            Console.WriteLine("Цикл while");
            int j = 20;
            while (j > 0)
            {
                Console.Write($"{digA}  ");
                if (j == 5) { Console.WriteLine(); }
                if (j == 9) { Console.WriteLine(); }
                if (j == 13) { Console.WriteLine(); }
                if (j == 17) { Console.WriteLine(); }   
                j--;
            }
            Console.WriteLine();
            Console.WriteLine("Цикл do..while");
            int d = 20;
            do
            {
                Console.Write($"{digA}  ");
                switch (d)
                {
                    case 5:
                        Console.WriteLine();
                        break;
                    case 9:
                        Console.WriteLine();
                        break;
                    case 13:
                        Console.WriteLine();
                        break;
                    case 17:
                        Console.WriteLine();
                        break;

                }
                d--;
            }
            while (d > 0);
           
            
            Console.ReadLine();
        }
    }
}
