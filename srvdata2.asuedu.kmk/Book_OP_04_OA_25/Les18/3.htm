Тема 3. Множества в языке C#.
Назад

Класс HashSet <T> в C# — это коллекция, которая содержит уникальные элементы и не имеет определенного порядка. Это часть пространства имен. A особенно полезен, когда вам нужно эффективно избегать дубликатов и быстро проверить наличие элементов.

  • HashSet <T> содержит неповторяющюся коллекцию элементов.
  • HashSet<T> содержит элементы неупорядоченно (без сортировки).
  • HashSet<T> позволяет быстро определить, есть такой элемент или нет (быстро потому что, использует индекс, который вычисляется из хэш-кода элемента).
  • HashSet<T> - имеет методы Add, Remove, Contains, но поскольку он использует хэш-реализацию, эти операции занимают 1 действие (методы Contains и Remove в List занимает n-действий.)

HashSet<T>  имеет методы:

  • UnionWith (объединение элементов с другим HashSet<T>)

  • IntersectWith (пересечение элементов с другим HashSet<T>)

  • ExceptWith (разность элементов с другим HashSet<T>)
  • SymmetricExceptWith (симетрическая разность элементов с другим HashSet<T>)

1. Инициализация:

Инициализировать объект можно различными способами:HashSet<T>

// Empty HashSet
HashSet<int> numbers = new HashSet<int>();

// HashSet from an array
int[] array = { 1, 2, 3, 4, 5 };
HashSet<int> setFromArray = new HashSet<int>(array);

2. Добавление элементов:

Чтобы добавить элемент в , используйте метод:HashSet<T>Add

numbers.Add(1);
numbers.Add(2);
numbers.Add(2); // Won't add since 2 is already present

3. Удаление элементов:

Чтобы удалить элемент, используйте метод:Remove

numbers.Remove(1);

4. Проверка на существование:

Чтобы проверить, существует ли элемент в , используйте метод:HashSet<T>Contains

bool containsTwo = numbers.Contains(2); // true

5. Перебор HashSet:

Вы можете выполнять итерации по циклу с использованием:HashSet<T>foreach

foreach (int num in numbers)
{
    Console.WriteLine(num);
}

6. Набор операций:

HashSet<T> Поддерживает различные операции с множествами, такие как объединение, пересечение и разность:

HashSet<int> setA = new HashSet<int> { 1, 2, 3, 4 };
HashSet<int> setB = new HashSet<int> { 3, 4, 5, 6 };

// Union
setA.UnionWith(setB);  (Объединение)
Console.WriteLine(string.Join(", ", setA)); // 1, 2, 3, 4, 5, 6

// Intersection   (Разность)
setA.IntersectWith(setB);
Console.WriteLine(string.Join(", ", setA)); // 3, 4, 5, 6

// Difference
setA.ExceptWith(setB);
Console.WriteLine(string.Join(", ", setA)); // 1, 2

7. Другие полезные методы:

  • Count: Возвращает количество элементов в .HashSet<T>

    int count = numbers.Count;

  • Clear: Удаляет все элементы из файла .HashSet<T>

    numbers.Clear();

  • IsSubsetOf и : Определяет, является ли a подмножеством или надмножеством указанной коллекции.IsSupersetOfHashSet<T>

    bool isSubset = setA.IsSubsetOf(setB);
bool isSuperset = setA.IsSupersetOf(setB);

Пример:

Давайте рассмотрим пример, в котором мы находим уникальные символы в строке с помощью :HashSet<T>

using System;
using System.Collections.Generic;

namespace HashSetExample
{
    class Program
    {
        static void Main()
        {
            string sentence = "Hello, world!";
            HashSet<char> uniqueChars = new HashSet<char>(sentence);

            // Removing punctuation and whitespace for clarity
            uniqueChars.Remove(',');
            uniqueChars.Remove(' ');
            uniqueChars.Remove('!');

            Console.WriteLine("Unique characters: " + string.Join(", ", uniqueChars));
        }
    }
}

На выходе получатся:

Unique characters: H, e, l, o, w, r, d

Обратите внимание, что порядок символов в выводе может отличаться, т.к. не гарантирует какого-либо конкретного порядка.HashSet<T>

 

using System;
using System.Collections.Generic; // подключаем HashSet
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace ConsoleApplication1
{
     class Program
     {
          static void Main(string[] args)
          {
               // создаем HashSet
               HashSet myHash = new HashSet();

               // добавляем элементы
               myHash.Add("a");
               myHash.Add("b");
               myHash.Add("c");
               // в myHash элементы: "a", "b", "c"

               // Перебираем все элементы с помощью foreach и выводим на экран
               foreach (string value in myHash)
                    Console.WriteLine(value);

               // узнать есть ли такой элемент?
               bool isFound = myHash.Contains("a");
               // isFound = true

               // удаляем элемент
               myHash.Remove("a");
               // в myHash элементы: "b", "c"

               // удаляем элементы "b" и "c"
               myHash.RemoveWhere(item => item=="b" || item=="c");
               // в myHash пусто

               // Метод ExceptWith (Разность)
               HashSet hashSet1 = new HashSet() { "a", "b", "c"};
               HashSet hashSet2 = new HashSet() { "a", "e", "h"};
               hashSet1.ExceptWith(hashSet2);
               // (ПОМЕНЯЛИСЬ) в hashSet1 элементы: "b", "c"
               // в hashSet2 элементы: "a", "e", "h"

               // Метод SymmetricExceptWith (Симетрическая разность)
               hashSet1 = new HashSet() { "a", "b", "c" };
               hashSet2 = new HashSet() { "a", "e", "h" };
               hashSet1.SymmetricExceptWith(hashSet2);
               // (ПОМЕНЯЛИСЬ) в hashSet1 элементы: "e", "b", "c", "h",
               // в hashSet2 элементы: "a", "e", "h"

               // Метод UnionWith (Объединение)
               hashSet1 = new HashSet() { "a", "b", "c" };
               hashSet2 = new HashSet() { "a", "e", "h" };
               hashSet1.UnionWith(hashSet2);
               // (ПОМЕНЯЛИСЬ) в hashSet1 элементы: "a", "b", "c", "e", "h",
               // в hashSet2 элементы: "a", "e", "h"
              
               // Метод IntersectWith (Пересечение)
               hashSet1 = new HashSet() { "a", "b", "c"};
               hashSet2 = new HashSet() { "a", "e", "h"};
               hashSet1.IntersectWith(hashSet2);
               // в hashSet1 элементы: "a"
               // в hashSet2 элементы: "a", "e", "h"

               // Метод Overlaps (есть ли Пересечение?)
               hashSet1 = new HashSet() { "a", "b", "c" };
               hashSet2 = new HashSet() { "a", "e", "h" };
               bool flag = hashSet1.Overlaps(hashSet2);
               // flag = true
               // в hashSet1 элементы: "a", "b", "c"
               // в hashSet2 элементы: "a", "e", "h"

               // Метод IsProperSubsetOf (является ли hashSet1 строгим подмножеством параметра HashSet)
               hashSet1 = new HashSet() { "a", "b"};
               flag = hashSet1.IsProperSubsetOf(new HashSet() { "a"}); // flag = false
               flag = hashSet1.IsProperSubsetOf(new HashSet() { "a", "b"}); // flag = false
               flag = hashSet1.IsProperSubsetOf(new HashSet() { "a", "b", "c" }); // flag = TRUE

               // Метод IsSubsetOf (является ли hashSet1 подмножеством параметра HashSet)
               hashSet1 = new HashSet() { "a", "b" };
               flag = hashSet1.IsSubsetOf(new HashSet() { "a" }); // flag = false
               flag = hashSet1.IsSubsetOf(new HashSet() { "a", "b" }); // flag = TRUE
               flag = hashSet1.IsSubsetOf(new HashSet() { "a", "b", "c" }); // flag = TRUE

               // Метод IsProperSupersetOf (является ли параметр HashSet строгим подмножеством hashSet1)
               hashSet1 = new HashSet() { "a", "b" };
               flag = hashSet1.IsProperSupersetOf(new HashSet() { "a" }); // flag = TRUE
               flag = hashSet1.IsProperSupersetOf(new HashSet() { "a", "b" }); // flag = false
               flag = hashSet1.IsProperSupersetOf(new HashSet() { "a", "b", "c" }); // flag = false

               // Метод IsSupersetOf (является ли параметр HashSet подмножеством hashSet1)
               hashSet1 = new HashSet() { "a", "b" };
               flag = hashSet1.IsSupersetOf(new HashSet() { "a" }); // flag = TRUE
               flag = hashSet1.IsSupersetOf(new HashSet() { "a", "b" }); // flag = TRUE
               flag = hashSet1.IsSupersetOf(new HashSet() { "a", "b", "c" }); // flag = false
          }
     }
}
