static void Main(string[] args)
        {
            Console.WriteLine("Передача массивов в качестве аргументов в C#.");
            //Передача одномерных массивов в качестве аргументов методам
            //1. объявляем и инициализируем массив
            int[] arr = { 1, 2, 3, 4 };
            // Объявляем метод  вывод на дисплей элементов массива
            void Result(int[] arrn)
            {
                // вывод на дисплей элементов массива
                for (int i = 0; i < arrn.Length; i++)
                {
                    Console.WriteLine("Элементы массива: " + arrn[i]);
                }
            }
            // Формирование функции для вывода массива
            int[] InMass (int n)
            {
                int[] newarray = new int[n];
                for (int i = 0; i < newarray.Length; i++)
                {
                    Console.Write($"Введите элемнте {i} массива  = ");
                    newarray[i] = int.Parse(Console.ReadLine());
                }
                return newarray;
            }

            Result(new int[] { 11, 12, 13, 14 });
            //Метод заполнения массива 
            Console.Write($"Введите количество элемнтов массива  = ");
            int nar = int.Parse(Console.ReadLine());  //количество элементов массива
            int[] resarray = new int[nar];  //Сформировали массив для результата

            Console.WriteLine($"Сформирован {nar} элементный массив");
            Result(resarray); //вывод сформированного массива

            Console.ReadLine();
        }
